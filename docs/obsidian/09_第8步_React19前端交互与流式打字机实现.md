## 1. 带 JWT Header 的 fetch-stream 解析器 (`ui/src/api/sse.ts`)

> [!IMPORTANT] 
> 原生 `EventSource` 无法携带自定义 Header，导致 JWT 只能暴露在 URL 参数中。本项目使用 `fetch` + `ReadableStream` 实现了安全可靠的 SSE 客户端。

```typescript
export async function streamSse(
  url: string,
  { signal, onOpen, onEvent }: StreamSseOptions,
): Promise<void> {
  const response = await fetchWithAuth(url, {
    method: "GET",
    headers: { Accept: "text/event-stream" },
    cache: "no-store",
    signal,
  });

  const reader = response.body!.getReader();
  const decoder = new TextDecoder();
  let buffer = "";

  try {
    while (!signal.aborted) {
      const { value, done } = await reader.read();
      if (done) break;
      
      buffer += decoder.decode(value, { stream: true });
      let boundary = buffer.search(/\r?\n\r?\n/);
      while (boundary !== -1) {
        const block = buffer.slice(0, boundary);
        buffer = buffer.slice(boundary + 2);
        const parsedEvent = parseEventBlock(block);
        if (parsedEvent) await onEvent(parsedEvent);
        boundary = buffer.search(/\r?\n\r?\n/);
      }
    }
  } finally {
    reader.releaseLock();
  }
}
```

## 2. 聊天打字机状态机 (`AgentChatView.tsx`)

使用 **“临时流式消息 (`stream:${sessionId}`) ➔ 数据库正式消息无缝替换”** 机制：

```typescript
const handleSseMessage = useCallback((sseMessage: SseMessage) => {
  const payloadMessage = sseMessage.payload?.message;
  const temporaryMessageId = `stream:${chatSessionId}`;

  // ① 收到文本分片：累加到临时消息 content 中（打字机效果）
  if (sseMessage.type === "AI_GENERATED_CONTENT_CHUNK") {
    setGenerating(true);
    setMessages((prev) => {
      const next = [...prev];
      const tempIndex = next.findIndex(m => m.id === temporaryMessageId);
      if (tempIndex >= 0) {
        next[tempIndex] = {
          ...next[tempIndex],
          content: next[tempIndex].content + payloadMessage.content,
        };
      } else {
        next.push({ ...payloadMessage, id: temporaryMessageId, role: "assistant" });
      }
      return next;
    });
    return;
  }

  // ② 收到完整持久化消息：替换为真实消息（含真实 ID 与 Token 统计）
  if (sseMessage.type === "AI_GENERATED_CONTENT") {
    setMessages((prev) => {
      const tempIndex = prev.findIndex(m => m.id === temporaryMessageId);
      if (tempIndex >= 0) {
        const next = [...prev];
        next[tempIndex] = payloadMessage; // 👈 替换
        return next;
      }
      return [...prev, payloadMessage];
    });
    return;
  }

  // ③ 状态卡片展示：思考中、调用工具中
  if (sseMessage.type === "AI_THINKING" || sseMessage.type === "AI_EXECUTING") {
    setDisplayAgentStatus(true);
    setAgentStatusText(sseMessage.payload?.statusText || "处理中");
    return;
  }

  // ④ 完成信号
  if (sseMessage.type === "AI_DONE") {
    setGenerating(false);
    setDisplayAgentStatus(false);
  }
}, [chatSessionId]);
```

## 3. 断线指数退避重连与历史对账

如果网络中断，自动执行指数退避重连（$1\text{s} \to 2\text{s} \to 4\text{s} \to 8\text{s} \to 15\text{s}$）：

```typescript
await waitForReconnect(
  Math.min(1000 * 2 ** Math.min(attempt - 1, 4), 15_000),
  controller.signal,
);
```
